Use Textual's `run_test()` harness, which drives the app headlessly through a `Pilot`, together with `pytest-asyncio`. Textual has no API for capturing posted messages. To test the message, mount your form in a small test app with a handler that records every `Submitted` message it receives, then assert on that list.

### 1. Setup

```toml
# pyproject.toml
[tool.pytest.ini_options]
asyncio_mode = "auto"   # without this, pytest-asyncio's strict mode rejects unmarked async tests
```

`pip install pytest-asyncio` (or add it to your dev dependencies).

### 2. Make the form testable

I don't have your code, so here's the shape I'm assuming. The parts that matter: the form posts its own message with a typed payload, and the error message has an id you can query.

```python
# myapp/form.py
from dataclasses import dataclass

from textual import on
from textual.app import ComposeResult
from textual.containers import VerticalGroup
from textual.message import Message
from textual.widgets import Button, Input, Label, Select

CATEGORIES = [("Bug", "bug"), ("Feature", "feature"), ("Question", "question")]


@dataclass(frozen=True)
class Entry:
    name: str
    category: str


class EntryForm(VerticalGroup):
    DEFAULT_CSS = """
    EntryForm #name-error { color: $error; display: none; }
    """

    class Submitted(Message):
        def __init__(self, entry: Entry) -> None:
            super().__init__()
            self.entry = entry

    def compose(self) -> ComposeResult:
        yield Input(placeholder="Name", id="name")
        yield Label("", id="name-error")
        yield Select(CATEGORIES, prompt="Category", id="category", allow_blank=False)
        yield Button("Submit", id="submit", variant="primary")

    @on(Button.Pressed, "#submit")
    def submit(self, event: Button.Pressed) -> None:
        event.stop()
        name_input = self.query_one("#name", Input)
        error = self.query_one("#name-error", Label)
        name = name_input.value.strip()
        if not name:
            error.update("Name is required")  # text, not just red, so it reads without color
            error.display = True
            name_input.focus()
            return
        error.display = False
        category = self.query_one("#category", Select).value
        self.post_message(self.Submitted(Entry(name, category)))
```

### 3. The tests

```python
# tests/test_entry_form.py
import pytest
from textual import on
from textual.app import App, ComposeResult
from textual.widgets import Input, Label, Select

from myapp.form import Entry, EntryForm


class FormHarness(App):
    """Mounts only the form and records every Submitted message."""

    def __init__(self) -> None:
        super().__init__()
        self.submitted: list[EntryForm.Submitted] = []

    def compose(self) -> ComposeResult:
        yield EntryForm()

    @on(EntryForm.Submitted)
    def record(self, message: EntryForm.Submitted) -> None:
        self.submitted.append(message)


@pytest.mark.parametrize("typed", ["", "   "])
async def test_blank_name_shows_error_and_posts_nothing(typed: str) -> None:
    app = FormHarness()
    async with app.run_test(size=(80, 24)) as pilot:
        app.query_one("#name", Input).value = typed
        await pilot.click("#submit")
        await pilot.pause()

        error = app.query_one("#name-error", Label)
        assert error.display
        assert str(error.content) == "Name is required"
        assert app.focused is app.query_one("#name", Input)
        assert app.submitted == []


async def test_submit_posts_entry() -> None:
    app = FormHarness()
    async with app.run_test(size=(80, 24)) as pilot:
        await pilot.click("#name")
        await pilot.press(*"Ada")                         # real keystrokes into the Input
        app.query_one("#category", Select).value = "feature"
        await pilot.click("#submit")
        await pilot.pause()                               # messages bubble asynchronously

        assert [m.entry for m in app.submitted] == [Entry("Ada", "feature")]
        assert not app.query_one("#name-error", Label).display


async def test_error_clears_after_fixing_name() -> None:
    app = FormHarness()
    async with app.run_test(size=(80, 24)) as pilot:
        await pilot.click("#submit")
        await pilot.pause()
        assert app.query_one("#name-error", Label).display

        await pilot.click("#name")
        await pilot.press(*"Ada")
        await pilot.click("#submit")
        await pilot.pause()

        assert not app.query_one("#name-error", Label).display
        assert len(app.submitted) == 1
```

**To test the full app without a test wrapper**, use `message_hook`. It sees every message the app handles:

```python
async def test_real_app_posts_entry() -> None:
    seen = []
    async with MyApp().run_test(message_hook=seen.append) as pilot:
        ...
        await pilot.pause()
    submitted = [m for m in seen if isinstance(m, EntryForm.Submitted)]
    assert [m.entry for m in submitted] == [Entry("Ada", "feature")]
```

### Gotchas

- **Call `await pilot.pause()` before asserting.** It waits for the app to finish handling the click and any messages it posted. Without it, the check can run first and the test fails intermittently.
- **Setting `Select.value` directly is the reliable approach.** Opening the dropdown with keystrokes depends on focus order and the overlay. If you need to prove the Select works from the keyboard, cover that in one separate test.
- **If you show errors with `self.notify(...)`, pass `run_test(notifications=True)`.** Headless runs turn toasts off by default, so an assertion on a notification would find nothing.
- **If you switch to `Input(validators=[...])`:** by default it validates on blur, on every change and on submit. That means the error can appear while the user is still typing. Set `validate_on=["submitted"]` (or `["blur"]`) to match the submit-only behavior above, and assert on `input.is_valid` or the `-invalid` class.
- **Textual 8 renamed `Select.BLANK` to `Select.NULL`.** If your Select allows a blank value and you check for "nothing chosen", compare against `Select.NULL`.
- **Snapshot tests of the error state need pinned versions.** For a visual check, `pytest-textual-snapshot`'s `snap_compare` works, but its last release needs `pytest>=8.4,<9`. On pytest 9, pip quietly installs an older version that writes `.raw` snapshots instead of `.svg`.

I haven't run these tests against your app. Check that the ids (`#name`, `#name-error`, `#category`, `#submit`) and the message class match what you actually have.
