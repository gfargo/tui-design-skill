## The form, written so it's easy to test

Two choices make testing easier. First, the validation rule is a plain function, so you can test it without a terminal. Second, the form posts one `Submitted` message carrying a value object, so a test can compare the whole thing with `==`.

```python
# myapp/entry_form.py
from dataclasses import dataclass

from textual import on
from textual.app import ComposeResult
from textual.containers import VerticalGroup
from textual.message import Message
from textual.widgets import Button, Input, Label, Select

CATEGORIES = [("Bug", "bug"), ("Feature", "feature"), ("Question", "question")]


@dataclass(frozen=True)
class Entry:
    name: str
    category: str


def validate_entry(name: str, category: str | None) -> dict[str, str]:
    errors: dict[str, str] = {}
    if not name.strip():
        errors["name"] = "Name is required"
    if category is None:
        errors["category"] = "Pick a category"
    return errors


class EntryForm(VerticalGroup):
    DEFAULT_CSS = """
    EntryForm .error { display: none; color: $error; }
    """

    class Submitted(Message):
        def __init__(self, entry: Entry) -> None:
            super().__init__()
            self.entry = entry

    def compose(self) -> ComposeResult:
        yield Label("Name")
        yield Input(id="name")
        yield Label("", id="name-error", classes="error")
        yield Select(CATEGORIES, id="category", prompt="Category")
        yield Label("", id="category-error", classes="error")
        yield Button("Submit", id="submit", variant="primary")

    @on(Button.Pressed, "#submit")
    @on(Input.Submitted, "#name")  # Enter in the name field submits too
    def submit(self) -> None:
        name = self.query_one("#name", Input)
        select = self.query_one("#category", Select)
        category = None if select.is_blank() else select.value
        errors = validate_entry(name.value, category)

        self._show_error("name", errors.get("name", ""))
        self._show_error("category", errors.get("category", ""))
        name.set_class("name" in errors, "-invalid")

        if errors:
            (name if "name" in errors else select).focus()
            return
        self.post_message(self.Submitted(Entry(name.value.strip(), category)))

    def _show_error(self, field: str, text: str) -> None:
        label = self.query_one(f"#{field}-error", Label)
        label.update(text)
        label.display = bool(text)
```

The error is shown as text, not just a red border, so it still makes sense in monochrome or with `NO_COLOR`.

## The tests

```python
# tests/test_entry_form.py
import pytest
from textual import on
from textual.app import App, ComposeResult
from textual.widgets import Input, Label, Select

from myapp.entry_form import Entry, EntryForm, validate_entry


class FormHarness(App[None]):
    """Mounts the form alone and records every Submitted message."""

    def __init__(self) -> None:
        super().__init__()
        self.submitted: list[Entry] = []

    def compose(self) -> ComposeResult:
        yield EntryForm()

    @on(EntryForm.Submitted)
    def record(self, event: EntryForm.Submitted) -> None:
        self.submitted.append(event.entry)


# Layer 1: the rule itself, no terminal needed
@pytest.mark.parametrize("name", ["", "   ", "\t"])
def test_blank_name_is_invalid(name):
    assert validate_entry(name, "bug") == {"name": "Name is required"}


def test_valid_entry_has_no_errors():
    assert validate_entry("Ada", "bug") == {}


# Layer 2: the widget, driven headlessly
@pytest.mark.parametrize("keys", [(), ("space", "space")])
async def test_empty_name_shows_error_and_posts_nothing(keys):
    app = FormHarness()
    async with app.run_test(size=(80, 24)) as pilot:
        app.query_one("#category", Select).value = "bug"
        await pilot.click("#name")
        await pilot.press(*keys)
        await pilot.click("#submit")
        await pilot.pause()  # let messages finish bubbling

        name_input = app.query_one("#name", Input)
        error = app.query_one("#name-error", Label)
        assert error.display
        assert str(error.content) == "Name is required"
        assert name_input.has_class("-invalid")
        assert app.focused is name_input  # focus goes back to the bad field
        assert app.submitted == []


async def test_submit_posts_entry():
    app = FormHarness()
    async with app.run_test(size=(80, 24)) as pilot:
        await pilot.click("#name")
        await pilot.press(*"Ada")
        app.query_one("#category", Select).value = "feature"
        await pilot.click("#submit")
        await pilot.pause()

        assert app.submitted == [Entry(name="Ada", category="feature")]
        assert not app.query_one("#name-error", Label).display


async def test_error_clears_after_fix():
    app = FormHarness()
    async with app.run_test(size=(80, 24)) as pilot:
        app.query_one("#category", Select).value = "bug"
        await pilot.click("#submit")
        await pilot.pause()
        assert app.query_one("#name-error", Label).display

        await pilot.click("#name")
        await pilot.press(*"Ada")
        await pilot.press("enter")  # also covers submitting from the keyboard
        await pilot.pause()

        assert not app.query_one("#name-error", Label).display
        assert app.submitted == [Entry(name="Ada", category="bug")]
```

## Things that will catch you out

- **Always `await pilot.pause()` before you assert.** Messages bubble asynchronously, so without it your recorder list may still be empty.
- **Setting `Select.value` directly is intentional.** Opening the dropdown and moving through it with keys depends on the order of the options. That makes tests fragile, and it tests Textual's widget rather than your code.
- **If your form is a `Screen`,** have the test app call `push_screen(EntryScreen())` in `on_mount`. Messages from the screen still bubble up to the app's `@on` handler.
- **Another way to capture messages:** `run_test(message_hook=...)` sees every message in the app. That's useful when you can't subclass the app, but you'll have to filter a lot of noise.
- **`run_test()` turns off notifications by default.** If you show errors with `self.notify(...)` instead of a label, pass `notifications=True` or your test will see nothing.
- **`pilot.click` only works on widgets visible at the test size.** If the form scrolls at 80×24, focus the button and use `pilot.press("enter")` instead.

## Optional: snapshot tests

`pytest-textual-snapshot` saves the rendered screen as an SVG and compares it on each run. For example, `snap_compare("myapp/app.py", press=[...])` captures the error state at 80×24 and at 60 columns. The catch: its latest release requires `pytest<9`. If you're on pytest 9, pip quietly installs an older version that writes `.raw` files instead of `.svg`. To use it, pin `pytest>=8.4,<9`. Otherwise, the tests above cover both of your requirements.
