Use Textual’s **`App.run_test()` + Pilot** to drive the form headlessly. Assert the error widget’s content and capture your custom submit message with `message_hook`.

Install async support for pytest:

```bash
python -m pip install pytest-asyncio
```

Here’s a minimal form showing the pattern. Adapt the IDs and message class to your app:

```python
# app.py
from textual import on
from textual.app import App, ComposeResult
from textual.message import Message
from textual.widgets import Button, Input, Label, Select


class EntryForm(App):
    class Submitted(Message):
        def __init__(self, name: str, role: str) -> None:
            super().__init__()
            self.name = name
            self.role = role

    def compose(self) -> ComposeResult:
        yield Input(placeholder="Name", id="name")
        yield Select(
            [("Reader", "reader"), ("Editor", "editor")],
            value="reader",
            allow_blank=False,
            id="role",
        )
        yield Label("", id="error")
        yield Button("Submit", id="submit")

    @on(Button.Pressed, "#submit")
    def submit(self) -> None:
        name_input = self.query_one("#name", Input)
        name = name_input.value.strip()
        error = self.query_one("#error", Label)

        if not name:
            error.update("Name is required")
            name_input.focus()
            return

        error.update("")
        role = self.query_one("#role", Select).value
        assert isinstance(role, str)
        self.post_message(self.Submitted(name, role))
```

The tests exercise both outcomes, including submission through the keyboard:

```python
# test_app.py
import pytest
from textual.widgets import Button, Input, Label, Select

from app import EntryForm


@pytest.mark.asyncio
@pytest.mark.parametrize("name", ["", "   "])
async def test_empty_name_shows_error_without_submitting(name):
    app = EntryForm()
    submitted = []

    def capture(message):
        if isinstance(message, EntryForm.Submitted):
            submitted.append(message)

    async with app.run_test(
        size=(80, 24), message_hook=capture
    ) as pilot:
        app.query_one("#name", Input).value = name

        assert await pilot.click("#submit")
        await pilot.pause()

        error = app.query_one("#error", Label)
        assert error.display
        assert error.visible
        assert str(error.content) == "Name is required"
        assert app.focused is app.query_one("#name", Input)
        assert submitted == []


@pytest.mark.asyncio
async def test_submit_posts_expected_message():
    app = EntryForm()
    submitted = []

    def capture(message):
        if isinstance(message, EntryForm.Submitted):
            submitted.append(message)

    async with app.run_test(
        size=(80, 24), message_hook=capture
    ) as pilot:
        await pilot.click("#name")
        await pilot.press(*"Ada")

        # Set the selection directly to isolate form submission behavior.
        app.query_one("#role", Select).value = "editor"
        await pilot.pause()

        app.query_one("#submit", Button).focus()
        await pilot.press("enter")
        await pilot.pause()

        assert [(m.name, m.role) for m in submitted] == [
            ("Ada", "editor")
        ]
        assert str(app.query_one("#error", Label).content) == ""
```

Run with `pytest -q`.

A few details matter:

- **Wait before asserting:** messages are processed asynchronously; use `await pilot.pause()` after interactions.
- **There’s no `app.messages` capture API:** use `message_hook`, as above, or a recorder handler on a test app subclass.
- **If the error uses `app.notify()`:** pass `notifications=True` to `run_test()`; notifications are disabled by default.
- These tests cover behavior. Add a snapshot test separately if you need to verify that the error remains readable at 80×24 and in a 60-column terminal.
