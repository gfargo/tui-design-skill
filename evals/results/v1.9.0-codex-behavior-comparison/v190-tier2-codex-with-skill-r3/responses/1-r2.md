Use Textual’s **`App.run_test()` and Pilot** to run the form headlessly. Assert the error widget’s state, and capture your custom message with a test-only handler.

Install the async pytest support:

```bash
python -m pip install pytest-asyncio
```

Here’s a complete example; adapt the IDs and message fields to your app.

**`form_app.py`**

```python
from textual import on
from textual.app import App, ComposeResult
from textual.message import Message
from textual.widgets import Button, Input, Label, Select


class FormApp(App):
    CSS = "#error { display: none; color: $error; }"

    class Submitted(Message):
        def __init__(self, name: str, role: str) -> None:
            super().__init__()
            self.name = name
            self.role = role

    def compose(self) -> ComposeResult:
        yield Input(placeholder="Name", id="name")
        yield Select(
            [("Reader", "reader"), ("Editor", "editor")],
            value="reader",
            allow_blank=False,
            id="role",
        )
        yield Label("Name is required", id="error")
        yield Button("Submit", id="submit")

    @on(Button.Pressed, "#submit")
    def submit(self) -> None:
        name_input = self.query_one("#name", Input)
        error = self.query_one("#error", Label)
        name = name_input.value.strip()

        error.display = not bool(name)
        if not name:
            name_input.focus()
            return

        role = self.query_one("#role", Select).value
        assert isinstance(role, str)
        self.post_message(self.Submitted(name, role))
```

**`test_form_app.py`**

```python
import pytest
from textual import on
from textual.widgets import Button, Input, Label, Select

from form_app import FormApp


class RecordingApp(FormApp):
    def __init__(self) -> None:
        super().__init__()
        self.submissions: list[FormApp.Submitted] = []

    @on(FormApp.Submitted)
    def record_submission(self, message: FormApp.Submitted) -> None:
        self.submissions.append(message)


@pytest.mark.asyncio
@pytest.mark.parametrize("name", ["", "   "])
async def test_empty_name_shows_error_without_submitting(name):
    app = RecordingApp()

    async with app.run_test(size=(80, 24)) as pilot:
        error = app.query_one("#error", Label)
        assert not error.display

        app.query_one("#name", Input).value = name
        await pilot.click("#submit")
        await pilot.pause()

        assert error.display
        assert str(error.content) == "Name is required"
        assert app.focused is app.query_one("#name", Input)
        assert app.submissions == []


@pytest.mark.asyncio
async def test_submit_posts_name_and_selected_role():
    app = RecordingApp()

    async with app.run_test(size=(80, 24)) as pilot:
        await pilot.click("#name")
        await pilot.press("A", "d", "a")

        # Set the value directly: this test checks the submitted payload.
        app.query_one("#role", Select).value = "editor"

        # Exercise keyboard activation of the submit button.
        app.query_one("#submit", Button).focus()
        await pilot.press("enter")
        await pilot.pause()

        assert not app.query_one("#error", Label).display
        assert len(app.submissions) == 1

        message = app.submissions[0]
        assert (message.name, message.role) == ("Ada", "editor")
```

Run with `pytest -q`.

The key details:

- `await pilot.pause()` lets queued messages and UI updates settle before assertions; avoid arbitrary sleeps.
- Textual has no `app.messages` capture API. A recorder handler, as above, observes actual message delivery.
- If your error uses `self.notify()` instead of an inline widget, pass `notifications=True` to `run_test()`—notifications are disabled by default.
- These behavioral tests don’t need screenshots. Add snapshots separately if you want to protect the form’s layout.
