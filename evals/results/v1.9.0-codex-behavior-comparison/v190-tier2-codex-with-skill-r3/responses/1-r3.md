Use Textual’s **`App.run_test()` and Pilot**. They run your app headlessly, drive inputs, and let you inspect widgets and messages.

Add async support to pytest:

```bash
python -m pip install pytest-asyncio
```

Here’s a test template. Replace the import, IDs, and message class with yours. This assumes:

- `#name` is an `Input`.
- `#role` is a `Select` with a value `"editor"`.
- `#error` is a `Static` showing validation text.
- `#submit` posts `Submitted(name=..., role=...)` when valid.

```python
# tests/test_entry_form.py
import pytest
from textual.widgets import Input, Select, Static

from myapp import EntryApp, Submitted


@pytest.mark.asyncio
@pytest.mark.parametrize("name", ["", "   "])
async def test_empty_name_shows_error(name):
    app = EntryApp()
    submitted = []

    def capture(message):
        if isinstance(message, Submitted):
            submitted.append(message)

    async with app.run_test(
        size=(80, 24),
        message_hook=capture,
    ) as pilot:
        app.query_one("#name", Input).value = name
        app.query_one("#role", Select).value = "editor"

        await pilot.click("#submit")
        await pilot.pause()

        error = app.query_one("#error", Static)
        assert error.display
        assert error.visible
        assert str(error.content) == "Name is required"
        assert submitted == []


@pytest.mark.asyncio
async def test_submit_posts_expected_message():
    app = EntryApp()
    submitted = []

    def capture(message):
        if isinstance(message, Submitted):
            submitted.append(message)

    async with app.run_test(
        size=(80, 24),
        message_hook=capture,
    ) as pilot:
        app.query_one("#name", Input).focus()
        await pilot.press("A", "d", "a")

        # Set the selection directly: this test checks the payload,
        # rather than Select's built-in keyboard navigation.
        app.query_one("#role", Select).value = "editor"
        await pilot.pause()

        # Exercise keyboard activation of the submit button.
        app.query_one("#submit").focus()
        await pilot.press("enter")
        await pilot.pause()

        assert len(submitted) == 1
        assert submitted[0].name == "Ada"
        assert submitted[0].role == "editor"
```

Your submit handler needs to perform validation explicitly—clicking a Button doesn’t automatically validate the form:

```python
@on(Button.Pressed, "#submit")
def submit(self) -> None:
    name = self.query_one("#name", Input).value.strip()
    error = self.query_one("#error", Static)

    if not name:
        error.update("Name is required")
        error.display = True
        self.query_one("#name", Input).focus()
        return

    error.display = False
    role = self.query_one("#role", Select).value
    self.post_message(Submitted(name=name, role=role))
```

Use `allow_blank=False` on the Select if a selection is mandatory, or validate its empty value too.

The key details are **`await pilot.pause()` before assertions**, because messages are asynchronous, and **`message_hook` to capture posted messages**—there is no `app.messages` collection. If your error uses a `notify()` toast instead of an inline widget, pass `notifications=True` to `run_test()`.
