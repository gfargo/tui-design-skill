Please review this Bubble Tea model before I merge it. Tell me what is wrong and how you would fix each problem. Rows come from a CI API, `Detail` can be a long free-text string, and people run this in small tmux panes.

```go
package main

const width = 80

type model struct {
	rows   []row
	ok     bool
	cursor int
}

func (m model) Init() tea.Cmd { return nil }

func (m model) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg := msg.(type) {
	case tea.KeyMsg:
		switch msg.String() {
		case "q", "ctrl+c":
			return m, tea.Quit
		case "r":
			resp, err := http.Get("https://ci.example.com/api/pipelines")
			if err != nil {
				m.ok = false
				return m, nil
			}
			defer resp.Body.Close()
			json.NewDecoder(resp.Body).Decode(&m.rows)
			m.ok = true
		}
	}
	return m, nil
}

func (m model) View() string {
	s := strings.Repeat("-", width) + "\n"
	for _, r := range m.rows {
		s += fmt.Sprintf("%-20s %s\n", r.Name, r.Detail)
	}
	color := lipgloss.Color("1")
	if m.ok {
		color = lipgloss.Color("2")
	}
	s += lipgloss.NewStyle().Foreground(color).Render("●")
	return s
}
```
